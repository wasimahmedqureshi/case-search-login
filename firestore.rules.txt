function csAdmin() {
  return request.auth != null
    && request.auth.token.email == 'wasimahmedqureshi@gmail.com'
    && request.auth.token.email_verified == true;
}
function csEmail() { return request.auth.token.email.lower(); }

match /cs_advocates/{id} {
  allow read: if request.auth != null && (csEmail() == id || csAdmin());
  allow write: if csAdmin();
}
match /cs_cases/{id} {
  allow read: if request.auth != null && (resource.data.owner == csEmail() || csAdmin());
  allow write: if csAdmin();
}
match /cs_history/{id} { allow read, write: if csAdmin(); }
match /cs_snap/{id}    { allow read, write: if csAdmin(); }
match /cs_admin/{id}   { allow read, write: if csAdmin(); }
